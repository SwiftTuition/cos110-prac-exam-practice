int Menu::numberAbove(string type, float value) {
    char code;
    if (type == "Starter") {
        code = 'S';
    } else if (type == "Main") {
        code = 'M';
    } else if (type == "Dessert") {
        code = 'D';
    } else {
        throw InvalidType(type);
    }
    int count = 0;
    for (int i = 0; i < maxSize; i++) {
        // The NULL check comes first: && stops at the first false part.
        if (items[i] != NULL && items[i]->getType() == code && items[i]->getPrice() > value) {
            count++;
        }
    }
    // No item is above the value.
    if (count == 0) {
        throw NothingAboveMinimumValue();
    }
    return count;
}

Menu& Menu::operator+=(Item* item) {
    // Use a free slot if there is one.
    for (int i = 0; i < maxSize; i++) {
        if (items[i] == NULL) {
            items[i] = item;
            return *this;
        }
    }
    // The menu is full: make an array with one more slot.
    Item** bigger = new Item*[maxSize + 1];
    for (int i = 0; i < maxSize; i++) {
        bigger[i] = items[i];
    }
    bigger[maxSize] = item;
    // Delete the old array of pointers only. The items now belong to the new array.
    delete[] items;
    items = bigger;
    maxSize++;
    return *this;
}

    // Invalid index (less than 0) or past the last bit: nothing to count.

        return 1 + countOnes(index + 1);

void BinaryStrings::insertAt(const Binary& b, int pos) {
    Node<Binary>* newNode = new Node<Binary>(b);
    // Front case: also covers the empty list.
    if (pos <= 0 || head == NULL) {
        newNode->next() = head;
        head = newNode;
        size++;
        return;
    }
    // Walk to the node before position pos, or to the last node.
    Node<Binary>* prev = head;
    int count = 1;
    while (prev->next() != NULL && count < pos) {
        prev = prev->next();
        count++;
    }
    // Link the new node in after prev.
    newNode->next() = prev->next();
    prev->next() = newNode;
    size++;
}

Finish with an explanation

Without notes, explain to a friend: why the NULL test comes first in a menu loop; the four steps of an assignment operator and what goes wrong without each; and why a recursive function’s base case must come before it reads a bit.

  • NULL first: an empty slot is NULL (Assumed). && stops at the first false part, so items[i] != NULL on the left keeps items[i]-> from reading through NULL.
  • Four steps: 1 self-assignment check (without it, a = a deletes what it is about to copy); 2 free the old memory (without it, a leak); 3 deep copy (without it, two objects share memory and it is deleted twice); 4 return *this; (without it, a = b = c does not work, and the build fails with -Werror, Assumed).
  • Base case first: the base case is the input with nothing left to read (index >= size, curr == NULL). If the function reads array[index] or curr->next() before that test, it reads past the end.
