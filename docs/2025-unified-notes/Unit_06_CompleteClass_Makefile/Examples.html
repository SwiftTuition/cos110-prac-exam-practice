Example 1: Basic Three-File Class

Let's start simple. We'll create a Temperature class that converts between Celsius and Fahrenheit. This shows the basic three-file pattern.

The Problem

Create a class that stores a temperature in Celsius and can convert it to Fahrenheit. The class should be split across three files with proper include guards.

File 1: temperature.h

// temperature.h - Header file
#ifndef TEMPERATURE_H
#define TEMPERATURE_H

class Temperature {
private:
    double celsius;
public:
    Temperature(double c);
    double getCelsius() const;
    double toFahrenheit() const;
    void setCelsius(double c);
};

#endif

File 2: temperature.cpp

// temperature.cpp - Implementation file
#include "temperature.h"

Temperature::Temperature(double c) {
    celsius = c;
}

double Temperature::getCelsius() const {
    return celsius;
}

double Temperature::toFahrenheit() const {
    return (celsius * 9.0 / 5.0) + 32.0;
}

void Temperature::setCelsius(double c) {
    celsius = c;
}

File 3: main.cpp

// main.cpp - Application file
#include "temperature.h"
#include <iostream>

int main() {
    Temperature room(22.0);

    std::cout << room.getCelsius() << "°C = ";
    std::cout << room.toFahrenheit() << "°F" << std::endl;

    room.setCelsius(0.0);
    std::cout << room.getCelsius() << "°C = ";
    std::cout << room.toFahrenheit() << "°F" << std::endl;

    return 0;
}

Makefile

CXX = g++
CXXFLAGS = -std=c++98 -Wall

program: main.o temperature.o
    $(CXX) $(CXXFLAGS) main.o temperature.o -o program

main.o: main.cpp temperature.h
    $(CXX) $(CXXFLAGS) -c main.cpp

temperature.o: temperature.cpp temperature.h
    $(CXX) $(CXXFLAGS) -c temperature.cpp

clean:
    rm -f *.o program

Example 2: Class with Dynamic Memory and Copy Constructor

This example shows a Playlist class that stores song names in a dynamic array. This requires a copy constructor, destructor, and assignment operator.

The Problem

Create a music playlist class that stores song titles. It needs to handle dynamic memory correctly to avoid leaks and ensure deep copying.

File 1: playlist.h

// playlist.h
#ifndef PLAYLIST_H
#define PLAYLIST_H

#include <string>

class Playlist {
private:
    std::string* songs;
    int capacity;
    int count;
public:
    Playlist(int cap);
    Playlist(const Playlist& other);
    ~Playlist();
    Playlist& operator=(const Playlist& other);

    void addSong(const std::string& title);
    void display() const;
    int getCount() const;
};

#endif

File 2: playlist.cpp

// playlist.cpp
#include "playlist.h"
#include <iostream>

Playlist::Playlist(int cap) {
    capacity = cap;
    count = 0;
    songs = new std::string[capacity];
}

Playlist::Playlist(const Playlist& other) {
    capacity = other.capacity;
    count = other.count;
    songs = new std::string[capacity];
    for (int i = 0; i < count; i++) {
        songs[i] = other.songs[i];
    }
}

Playlist::~Playlist() {
    delete[] songs;
}

Playlist& Playlist::operator=(const Playlist& other) {
    if (this == &other) {
        return *this;
    }

    delete[] songs;

    capacity = other.capacity;
    count = other.count;
    songs = new std::string[capacity];
    for (int i = 0; i < count; i++) {
        songs[i] = other.songs[i];
    }

    return *this;
}

void Playlist::addSong(const std::string& title) {
    if (count < capacity) {
        songs[count] = title;
        count++;
    }
}

void Playlist::display() const {
    std::cout << "Playlist (" << count << " songs):" << std::endl;
    for (int i = 0; i < count; i++) {
        std::cout << (i + 1) << ". " << songs[i] << std::endl;
    }
}

int Playlist::getCount() const {
    return count;
}

File 3: main.cpp

// main.cpp
#include "playlist.h"
#include <iostream>

int main() {
    Playlist myMusic(5);

    myMusic.addSong("Eloko Oyo - Fally Ipupa");
    myMusic.addSong("Original - Fally Ipupa");
    myMusic.addSong("Juste Une Fois - Fally Ipupa");

    myMusic.display();

    Playlist backup = myMusic;
    std::cout << "\nBackup created. Backup has " << backup.getCount() << " songs." << std::endl;

    return 0;
}

Example 3: Complete Class with Namespace and Makefile

This example shows a complete student record system with namespaces, operator overloading, and a proper makefile. This is production-quality code.

The Problem

Create a student record system that stores student information (name, student number, marks). Use namespaces to organize the code and overload the stream insertion operator for easy output.

File 1: student.h

// student.h
#ifndef STUDENT_H
#define STUDENT_H

#include <string>
#include <iostream>

namespace UniversitySystem {
    class Student {
    private:
        std::string name;
        std::string studentNumber;
        double mark;
    public:
        Student();
        Student(const std::string& n, const std::string& num, double m);

        std::string getName() const;
        std::string getStudentNumber() const;
        double getMark() const;

        void setMark(double m);

        friend std::ostream& operator<<(std::ostream& out, const Student& s);
    };
}

#endif

File 2: student.cpp

// student.cpp
#include "student.h"

namespace UniversitySystem {
    Student::Student() {
        name = "Unknown";
        studentNumber = "u00000000";
        mark = 0.0;
    }

    Student::Student(const std::string& n, const std::string& num, double m) {
        name = n;
        studentNumber = num;
        mark = m;
    }

    std::string Student::getName() const {
        return name;
    }

    std::string Student::getStudentNumber() const {
        return studentNumber;
    }

    double Student::getMark() const {
        return mark;
    }

    void Student::setMark(double m) {
        mark = m;
    }

    std::ostream& operator<<(std::ostream& out, const Student& s) {
        out << s.studentNumber << ": " << s.name;
        out << " - " << s.mark << "%";
        if (s.mark >= 50) {
            out << " (Pass)";
        } else {
            out << " (Fail)";
        }
        return out;
    }
}

File 3: main.cpp

// main.cpp
#include "student.h"
#include <iostream>

using UniversitySystem::Student;

int main() {
    Student s1("Naledi Khumalo", "u21234567", 78.5);
    Student s2("Thabo Mokoena", "u21456789", 45.0);
    Student s3("Zanele Dlamini", "u21987654", 62.0);

    std::cout << "COS 110 Class List:" << std::endl;
    std::cout << s1 << std::endl;
    std::cout << s2 << std::endl;
    std::cout << s3 << std::endl;

    s2.setMark(52.0);
    std::cout << "\nAfter supplementary exam:" << std::endl;
    std::cout << s2 << std::endl;

    return 0;
}

Makefile

CXX = g++
CXXFLAGS = -std=c++98 -Wall

program: main.o student.o
    $(CXX) $(CXXFLAGS) main.o student.o -o program

main.o: main.cpp student.h
    $(CXX) $(CXXFLAGS) -c main.cpp

student.o: student.cpp student.h
    $(CXX) $(CXXFLAGS) -c student.cpp

clean:
    rm -f *.o program

Example 4: Find the Bug

This code has several bugs related to separate compilation. Your task is to find and fix them. The code won't compile or link correctly.

The Problem

A student tried to create a Counter class but made several mistakes. Find all the bugs.

File 1: counter.h

// counter.h
class Counter {
private:
    int count;
public:
    Counter(int initial);
    void increment();
    void decrement();
    int getValue();
};

File 2: counter.cpp

// counter.cpp
#include "counter.h"

void Counter::Counter(int initial) {
    count = initial;
}

void increment() {
    count++;
}

void Counter::decrement() {
    count--;
}

int Counter::getValue() {
    return count;
}

File 3: main.cpp

// main.cpp
#include "counter.cpp"
#include <iostream>

int main() {
    Counter c(10);
    c.increment();
    c.increment();
    std::cout << "Count: " << c.getValue() << std::endl;
    return 0;
}

Bugs Found

Bug 1 (counter.h): Missing include guards! If this header gets included multiple times, you'll get redefinition errors.
Bug 2 (counter.h): getValue() should be const since it doesn't modify the object.
Bug 3 (counter.cpp): Constructor definition is wrong. void Counter::Counter(int initial) should be Counter::Counter(int initial) (no return type).
Bug 4 (counter.cpp): increment() is missing the Counter:: scope resolution operator. It's being defined as a standalone function instead of a member function.
Bug 5 (main.cpp): Including counter.cpp instead of counter.h. This will cause multiple definition errors when linking.

Corrected Code

File 1: counter.h (Fixed)

// counter.h - FIXED
#ifndef COUNTER_H
#define COUNTER_H

class Counter {
private:
    int count;
public:
    Counter(int initial);
    void increment();
    void decrement();
    int getValue() const;
};

#endif

File 2: counter.cpp (Fixed)

// counter.cpp - FIXED
#include "counter.h"

Counter::Counter(int initial) {
    count = initial;
}

void Counter::increment() {
    count++;
}

void Counter::decrement() {
    count--;
}

int Counter::getValue() const {
    return count;
}

File 3: main.cpp (Fixed)

// main.cpp - FIXED
#include "counter.h"
#include <iostream>

int main() {
    Counter c(10);
    c.increment();
    c.increment();
    std::cout << "Count: " << c.getValue() << std::endl;
    return 0;
}
